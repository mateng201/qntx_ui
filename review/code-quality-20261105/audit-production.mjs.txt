import fs from "node:fs";
import path from "node:path";
import http from "node:http";
import { chromium } from "@playwright/test";

const root = path.resolve("apps/web/dist");
const server = http.createServer((request, response) => {
  const file = path.resolve(root, `.${new URL(request.url, "http://localhost").pathname === "/" ? "/index.html" : new URL(request.url, "http://localhost").pathname}`);
  if (!file.startsWith(`${root}${path.sep}`) || !fs.existsSync(file)) { response.writeHead(404); response.end(); return; }
  response.setHeader("Content-Type", file.endsWith(".js") ? "text/javascript" : file.endsWith(".css") ? "text/css" : file.endsWith(".svg") ? "image/svg+xml" : "text/html");
  response.end(fs.readFileSync(file));
});
await new Promise((resolve) => server.listen(0, "127.0.0.1", resolve));
const browser = await chromium.launch({ channel: "chrome" });
try {
  const page = await browser.newPage();
  const javascript = [];
  page.on("request", (request) => { if (request.url().endsWith(".js")) javascript.push(new URL(request.url()).pathname); });
  await page.goto(`http://127.0.0.1:${server.address().port}/`);
  await page.waitForFunction(() => document.querySelector("[data-home-page]")?.dataset.scrollReady === "true");
  const native = ["android", "ios"].map((platform) => {
    const dir = `apps/mobile/dist/_expo/static/js/${platform}`;
    const file = fs.readdirSync(dir).find((name) => name.endsWith(".hbc"));
    const buffer = fs.readFileSync(path.join(dir, file));
    return { platform, forbiddenPathMarkers: ["apps/web/", "apps\\web\\", "@astrojs/", "astro/dist/", "home-page-scroll", "wheel-section-navigation"].filter((marker) => buffer.includes(Buffer.from(marker))) };
  });
  const result = { productionJavascriptRequests: javascript, nativeBinaryMarkerScan: native, limitation: "Native binary marker absence is supplemental; source boundary rules and imports provide the primary evidence. No complete native source map was collected." };
  fs.writeFileSync("review/code-quality-20261105/evidence/production-analysis.json", JSON.stringify(result, null, 2) + "\n");
  console.log(JSON.stringify(result, null, 2));
} finally {
  await browser.close();
  await new Promise((resolve) => server.close(resolve));
}
